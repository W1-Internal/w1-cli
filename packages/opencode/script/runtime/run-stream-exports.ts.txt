
export { serve };

export function installRuntimeSignalHandlers(): () => void {
  const interruptActiveRun = () => {
    if (activeRunAbort) {
      activeRunAbort.abort("Interrupted by user.");
      return;
    }
    void terminateInProcessJobs("interrupted: W1 closed while this job was running").finally(() =>
      exitAfterFlush(0),
    );
  };
  process.on("SIGTERM", interruptActiveRun);
  process.on("SIGINT", interruptActiveRun);
  return () => {
    process.off("SIGTERM", interruptActiveRun);
    process.off("SIGINT", interruptActiveRun);
  };
}
