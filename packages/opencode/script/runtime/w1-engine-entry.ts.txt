import { existsSync, readFileSync } from "node:fs";
import path from "node:path";
import { fileURLToPath } from "node:url";

import { W1FileAuthCoordinator } from "../../src/engine/auth.ts";
import { W1EngineError, W1_ENGINE_PROTOCOL_VERSION } from "../../src/engine/contracts.ts";
import { W1Engine } from "../../src/engine/core.ts";
import { W1EngineServer } from "../../src/engine/local-ipc.ts";
import { stdioThreadWorkerFactory, W1WorkerSupervisor } from "../../src/engine/worker-supervisor.ts";

const valueAfter = (flag: string): string | undefined => {
  const index = process.argv.indexOf(flag);
  return index >= 0 ? process.argv[index + 1] : undefined;
};

const scriptDirectory = path.dirname(fileURLToPath(import.meta.url));
const workerCommand = valueAfter("--worker-command") || process.execPath;
const workerArgs = process.argv.flatMap((value, index) =>
  value === "--worker-arg" && process.argv[index + 1] ? [process.argv[index + 1]!] : []);
if (!workerArgs.length) throw new Error("The packaged W1 Engine requires explicit worker arguments.");

const packageVersion = valueAfter("--package-version") || "0.0.0";
const buildFile = path.join(scriptDirectory, "BUILD_ID");
const buildId = existsSync(buildFile) ? readFileSync(buildFile, "utf8").trim() : "missing";
const configuredBackendUrl = valueAfter("--backend-url") || process.env.W1_BACKEND_URL || "https://app.w1lab.com";
const backendUrl = (() => {
  const parsed = new URL(configuredBackendUrl);
  if (parsed.username || parsed.password || !["http:", "https:"].includes(parsed.protocol)) {
    throw new Error("The W1 backend URL is invalid.");
  }
  const local = ["localhost", "127.0.0.1", "::1"].includes(parsed.hostname);
  if (!local && parsed.hostname !== "w1lab.com" && !parsed.hostname.endsWith(".w1lab.com")) {
    throw new Error("The W1 backend URL is outside the trusted W1 domains.");
  }
  return parsed.toString().replace(/\/$/, "");
})();

const engine = new W1Engine();
const supervisor = new W1WorkerSupervisor(engine, stdioThreadWorkerFactory({
  command: workerCommand,
  args: workerArgs,
  env: {
    ...(process.env.ELECTRON_RUN_AS_NODE === "1" ? { ELECTRON_RUN_AS_NODE: "1" } : {}),
    W1_BACKEND_URL: backendUrl,
    WILLIAMS_ONE_PROVIDER: "w1",
    W1_SURFACE: "engine",
  },
}), 2);
const server = new W1EngineServer({
  engine,
  supervisor,
  auth: new W1FileAuthCoordinator(),
  buildId,
  minimumClientVersion: process.env.W1_MIN_CLIENT_VERSION || "0.0.0",
});

const activeCount = () => (supervisor as unknown as { active: Map<string, unknown> }).active.size;
let closing = false;
let stopWhenIdle = false;
async function close(signal: string): Promise<void> {
  if (closing) return;
  closing = true;
  process.stderr.write(`[w1-engine] stopping (${signal})\n`);
  try {
    await server.close();
    process.exitCode = 0;
  } catch {
    process.exitCode = 1;
  }
}

type InternalServer = {
  dispatch(request: { method: string; params?: unknown }, socket: unknown): Promise<unknown>;
};
const controlled = server as unknown as InternalServer;
const dispatch = controlled.dispatch.bind(server);
controlled.dispatch = async (request, socket) => {
  if (request.method === "engine.status") {
    const activeThreads = activeCount();
    return {
      pid: process.pid,
      protocolVersion: W1_ENGINE_PROTOCOL_VERSION,
      buildId,
      packageVersion,
      activeThreads,
      idle: activeThreads === 0,
      stopWhenIdle,
    };
  }
  if (request.method === "engine.stop") {
    const params = request.params && typeof request.params === "object" ? request.params as Record<string, unknown> : {};
    const activeThreads = activeCount();
    if (activeThreads > 0 && params.whenIdle !== true) {
      throw new W1EngineError("engine", "W1 Engine is busy and refused to stop.", {
        retryable: true,
        structural: { reason: "engine_busy", activeThreads },
      });
    }
    stopWhenIdle = activeThreads > 0;
    if (!stopWhenIdle) setImmediate(() => { void close("client request"); });
    return { accepted: true, stopping: !stopWhenIdle, whenIdle: stopWhenIdle, activeThreads };
  }
  return dispatch(request, socket);
};

const idlePoll = setInterval(() => {
  if (stopWhenIdle && activeCount() === 0) void close("idle upgrade");
}, 250);
idlePoll.unref();

await server.listen();
process.stderr.write(`[w1-engine] ready on ${server.endpoint}\n`);
process.once("SIGINT", () => { void close("SIGINT"); });
process.once("SIGTERM", () => { void close("SIGTERM"); });
